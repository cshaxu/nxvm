add_library(myob300-board STATIC ob300_board.c ob300_card.c ob300_chipset.c ob300_display.c ob300_mouse.c)
target_include_directories(myob300-board PUBLIC "${CMAKE_SOURCE_DIR}/src")
target_link_libraries(myob300-board PUBLIC x86-nxvm-board-base PRIVATE x86-nxvm-chip-ct-vga)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(myob300-board PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_library(myob300-machine STATIC ob300_machine.c)

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(myob300-machine PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

target_link_libraries(myob300 PRIVATE myob300-machine x86-nxvm-product)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(myob300 PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(myob300 PRIVATE -Wl,--strip-debug)
endif()
