foreach(profile IN ITEMS 80186 80286 80386)
    # Retain the existing two output contracts: package-local decoder output
    # and repository timing-ledger input. Both use the same CPU-owned program.
    foreach(kind IN ITEMS decoder inventory)
        set(target "x86-core-${kind}-${profile}")
        add_executable(${target} "cpu/cpu_${profile}_decoder_inventory_runner.c")
        target_link_libraries(${target} PRIVATE x86-core-cpu)
        if(MSVC)
            target_compile_options(${target} PRIVATE /UNDEBUG)
        else()
            target_compile_options(${target} PRIVATE -UNDEBUG)
        endif()
        if(kind STREQUAL "inventory")
            set(output "${CMAKE_BINARY_DIR}/generated/test-results/${profile}-decoder-inventory.json")
        else()
            set(output "${CMAKE_CURRENT_BINARY_DIR}/decoder-${profile}.json")
        endif()
        file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated/test-results")
        target_compile_definitions(${target} PRIVATE PROJECT_TEST_${profile}_DECODER_PATH="${output}")
        add_test(NAME "x86-core.cpu/${kind}-${profile}" COMMAND ${target})
        set_tests_properties("x86-core.cpu/${kind}-${profile}" PROPERTIES LABELS "unit;x86-core" TIMEOUT 30)
        list(APPEND SHARED_X86_CORE_TEST_TARGETS ${target})
    endforeach()
endforeach()

foreach(check IN ITEMS address_span decode_admission host_arithmetic stack_frame)
    string(REPLACE "_" "-" name "${check}")
    add_test(NAME "x86-core.cpu/${name}-boundary" COMMAND "${CMAKE_COMMAND}"

    set_tests_properties("x86-core.cpu/${name}-boundary" PROPERTIES LABELS "unit;x86-core" TIMEOUT 30)
endforeach()
