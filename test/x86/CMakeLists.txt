cmake_minimum_required(VERSION 3.23)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(shared_x86_tests LANGUAGES C)
    enable_testing()
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
get_filename_component(X86_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../src/x86" ABSOLUTE)
get_filename_component(COMMON_ROOT "${X86_ROOT}/../common" ABSOLUTE)
if(NOT TARGET x86-debug)
    add_subdirectory("${X86_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/x86")
endif()

function(x86_test source)
    get_filename_component(name "${source}" NAME)
    add_executable(x86-test-${name} ${source}_smoke.c)
    target_link_libraries(x86-test-${name} PRIVATE ${ARGN})
    if(MSVC)
        target_compile_options(x86-test-${name} PRIVATE /UNDEBUG)
    else()
        target_compile_options(x86-test-${name} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME x86.${name} COMMAND x86-test-${name})
    set_tests_properties(x86.${name} PROPERTIES LABELS "unit;x86"
        TIMEOUT 30 WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
endfunction()

endif()
