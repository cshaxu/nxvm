function(mynes_add_machine_smoke name source)
    string(REPLACE ".c" "" test_name "${source}")
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE mynes-machine-driver)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${name} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME mynes.machine.${test_name} COMMAND ${name})
endfunction()

mynes_add_machine_smoke(mynes-machine-owner-rom-probe owner_rom_probe.c)
mynes_add_machine_smoke(mynes-machine-pacing-smoke pacing_smoke.c)
mynes_add_machine_smoke(mynes-machine-battery-persistence-smoke battery_persistence_smoke.c)
mynes_add_machine_smoke(mynes-machine-media-title-suffix-smoke media_title_suffix_smoke.c)
mynes_add_machine_smoke(mynes-machine-media-failure-contract-smoke media_failure_contract_smoke.c)
mynes_add_machine_smoke(mynes-machine-palette-smoke palette_smoke.c)
