foreach(variant IN ITEMS tft dstn)
    if(TARGET mypc110-product-${variant})
        add_executable(mypc110-${variant}-product-composition-smoke
            unit/product_composition_smoke.c)
        target_link_libraries(mypc110-${variant}-product-composition-smoke PRIVATE
            mypc110-product-${variant})
        if(variant STREQUAL "tft")
            target_compile_definitions(mypc110-${variant}-product-composition-smoke PRIVATE
                MYPC110_EXPECTED_MACHINE_NAME="ibm-pc110-tft"
                MYPC110_EXPECTED_PANEL_VALUE=0x00u)
        else()
            target_compile_definitions(mypc110-${variant}-product-composition-smoke PRIVATE
                MYPC110_EXPECTED_MACHINE_NAME="ibm-pc110-dstn-derived"
                MYPC110_EXPECTED_PANEL_VALUE=0x15u)
        endif()
        if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(mypc110-${variant}-product-composition-smoke PRIVATE
                -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
        endif()
        add_dependencies(mypc110-${variant}-product-composition-smoke
            mypc110-${variant}-preview)
        if(TARGET build-unit-tests)
            add_dependencies(build-unit-tests mypc110-${variant}-product-composition-smoke)
        endif()
        add_test(NAME mypc110.unit/${variant}-product-composition
            COMMAND mypc110-${variant}-product-composition-smoke)
        set_tests_properties(mypc110.unit/${variant}-product-composition PROPERTIES
            LABELS "unit;mypc110" TIMEOUT 30)
    endif()
endforeach()

if(TARGET mypc110-real-firmware-tft)
    add_executable(mypc110-real-firmware-post-integration
        integration/real_firmware_post_integration.c)
    target_link_libraries(mypc110-real-firmware-post-integration PRIVATE
        mypc110-board mypc110-real-firmware-tft)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(mypc110-real-firmware-post-integration PRIVATE
            -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
    endif()
    # One unchanged-BIOS receiver covers the earlier timer/SMI prefixes and
    # the integrated controller's reset, IRQ and four SENSE results.
    add_test(NAME mypc110.integration/real-firmware-fdc-reset
        COMMAND mypc110-real-firmware-post-integration --fdc-reset)
    set_tests_properties(mypc110.integration/real-firmware-fdc-reset PROPERTIES
        LABELS "integration;mypc110" TIMEOUT 180)
    # Retained power plus ordinary POST/INT19, through one real BIOS replay.
    # Keep this CPU-bound whole route isolated from other CTest workloads.
    add_test(NAME mypc110.integration/real-firmware-boot-handoff
        COMMAND mypc110-real-firmware-post-integration --resume-boot)
    # The x86 host executes the unchanged guest much more slowly; retain a
    # bounded budget above the measured successful 719-second handoff.
    if(CMAKE_SIZEOF_VOID_P EQUAL 4)
        set(mypc110_boot_timeout 1080)
    else()
        set(mypc110_boot_timeout 360)
    endif()
    set_tests_properties(mypc110.integration/real-firmware-boot-handoff PROPERTIES
        LABELS "integration;mypc110" TIMEOUT ${mypc110_boot_timeout} RUN_SERIAL TRUE)
endif()
